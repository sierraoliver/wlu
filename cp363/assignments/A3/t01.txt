SELECT m.memberSurname, m.memberForename, 
(SELECT COUNT(*) FROM pub WHERE pubMemberId = m.memberId AND pubPubType = 'b') AS books,
(SELECT COUNT(*) FROM pub WHERE pubMemberId = m.memberId AND pubPubType = 'a') AS articles,
(SELECT COUNT(*) FROM pub WHERE pubMemberId = m.memberId AND pubPubType = 'p') AS papers
FROM member AS m
ORDER BY m.memberSurname, m.memberForename
