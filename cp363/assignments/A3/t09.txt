SELECT m.memberSurname, m.memberForename,
(SELECT COUNT(*) 
 FROM memberNarrow
 WHERE memberNarrowMemberId = m.memberId
) AS narrowCount
FROM member AS m
WHERE (SELECT COUNT(*) 
       FROM memberNarrow
       WHERE memberNarrowMemberId = m.memberId
) >= 30
ORDER BY m.memberSurname, m.memberForename
