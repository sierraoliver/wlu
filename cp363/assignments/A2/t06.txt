SELECT member.memberSurname, member.memberForename, COUNT(CASE WHEN pubType.pubTypeDesc = 'Book' THEN 1 END) AS numPublications FROM member 
LEFT JOIN pub ON member.memberId = pub.pubMemberId
LEFT JOIN pubType ON pub.pubPubType = pubType.pubType 
GROUP BY member.memberSurname, member.memberForename
ORDER BY member.memberSurname, member.memberForename
